# ==============================================================================
# DELTO-3F IN-HAND MANIPULATION: COMMAND CHEAT SHEET
# ==============================================================================

# ------------------------------------------------------------------------------
# 1. INTERACTIVE 3D VISUAL EVALUATION & DEMONSTRATIONS
# ------------------------------------------------------------------------------
# A. Interactive Launcher (One-Click Menu)
bash scripts/run_eval_demo.sh

# B. Nominal Clean Presentation Viewer (Ideal kinematics for video capture)
python3 src/evaluation/evaluate.py \
  --checkpoint models/best_model_clean.pth \
  --render \
  --hold_time 1.5

# C. Sim-to-Real Domain Randomization Viewer (Real-time HUD, sensor noise, latency)
python3 src/evaluation/evaluate_dr.py \
  --checkpoint models/best_model_with_dr.pth \
  --render \
  --smooth_alpha 0.50

# ------------------------------------------------------------------------------
# 2. HIGH-THROUGHPUT PARALLEL BENCHMARKING
# ------------------------------------------------------------------------------
# A. 16-Worker Stressed Sim-to-Real Benchmark (600 stratified episodes)
python3 src/evaluation/evaluate_dr.py \
  --checkpoint models/best_model_with_dr.pth \
  --episodes_per_range 100 \
  --num_workers 16 \
  --smooth_alpha 0.50 \
  --epsilon_ratio 0.10 \
  --report_file evaluation/reports/EVALUATION_DR_REPORT.md

# B. 16-Worker Clean Nominal Baseline Benchmark (600 stratified episodes)
python3 src/evaluation/evaluate.py \
  --checkpoint models/best_model_clean.pth \
  --episodes_per_range 100 \
  --num_workers 16 \
  --epsilon_ratio 0.10

# C. Dedicated benchmark script runner
bash scripts/run_benchmark.sh

# ------------------------------------------------------------------------------
# 3. TELEOPERATION, DEMONSTRATION & DATA PIPELINE
# ------------------------------------------------------------------------------
# A. Collect raw human demonstrations via webcam ArUco tracking
python3 src/teleoperation/collect_data.py

# B. Inspect quality of collected demonstrations (drop detection & action jitter)
python3 src/teleoperation/inspect_data.py data/processed

# C. Replay demonstration trajectories in MuJoCo viewer
python3 src/teleoperation/replay_data.py

# D. Preprocess raw demonstrations (trim dead time, compute smooth velocities)
python3 src/teleoperation/preprocess_data.py \
  --input_dir data/raw \
  --output_dir data/processed

# ------------------------------------------------------------------------------
# 4. POLICY TRAINING & REINFORCEMENT LEARNING
# ------------------------------------------------------------------------------
# A. Pre-train Behavior Cloning (BC) Actor Network from demonstrations
python3 src/training/train_bc.py

# B. Launch Deep PPO Reinforcement Learning with Domain Randomization
python3 src/training/train_ppo.py \
  --pretrained_actor models/best_model_with_dr.pth \
  --num_envs 16 \
  --smooth_alpha 0.50 \
  --sim2real_dr \
  --action_delay_max 1 \
  --vision_delay_max 2 \
  --obs_noise_scale 0.80 \
  --dr_ratio 0.30 \
  --output_dir experiments/dr_finetune_run1

# C. Failure Mode Diagnostic Profiler
python3 src/evaluation/diagnose_failures.py
